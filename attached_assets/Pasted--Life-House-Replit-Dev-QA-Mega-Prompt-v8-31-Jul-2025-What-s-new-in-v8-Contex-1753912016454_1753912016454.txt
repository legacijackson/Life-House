## 2  DB Schema Δ (Support FAQ)

```sql
-- master FAQ articles
CREATE TABLE faqs (
  id          serial PRIMARY KEY,
  question    text NOT NULL,
  answer      text NOT NULL,
  weight      int  DEFAULT 100,
  created_at  timestamptz DEFAULT now()
);
-- link an FAQ to one or more app pages (by route path pattern)
CREATE TABLE faq_pages (
  faq_id int REFERENCES faqs(id) ON DELETE CASCADE,
  path_pattern text  -- e.g. '/portal*' or '/admin/*'
);
-- link an FAQ to one or more roles
CREATE TABLE faq_roles (
  faq_id int REFERENCES faqs(id) ON DELETE CASCADE,
  role   user_role_enum
);
-- feedback table
CREATE TABLE faq_feedback (
  id serial PRIMARY KEY,
  faq_id int REFERENCES faqs(id),
  user_id int REFERENCES users(id),
  helpful boolean,
  created_at timestamptz DEFAULT now()
);
```
